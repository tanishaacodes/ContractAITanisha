# models/caselaw_bert.py
from transformers import AutoTokenizer, AutoModel
import torch
class CaseLawBERT:
    def __init__(self, model_name="nlpaueb/legal-bert-base-uncased"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name)
    def embed(self, text):
        inputs = self.tokenizer(
            text,
            return_tensors="pt",
            truncation=True,
            padding=True,
            max_length=512
        )
        with torch.no_grad():
            outputs = self.model(**inputs)
        # Mean pooling
        embeddings = outputs.last_hidden_state.mean(dim=1)
        return embeddings.squeeze().numpy()

# services/case_index.py
import faiss
import numpy as np
class CaseLawIndex:
    def __init__(self, dim=768):
        self.index = faiss.IndexFlatL2(dim)
        self.metadata = []
    def add_case(self, embedding, case_text, case_id):
        self.index.add(np.array([embedding]).astype("float32"))
        self.metadata.append({
            "case_id": case_id,
            "text": case_text
        })
    def search(self, query_embedding, k=5):
        D, I = self.index.search(
            np.array([query_embedding]).astype("float32"), k
        )
        results = []
        for idx in I[0]:
            results.append(self.metadata[idx])
        return results

from services.case_retrieval import CaseRetriever
from neo4j import GraphDatabase
class RAGRetriever:
    def __init__(self):
        self.case_retriever = CaseRetriever()
        self.graph = GraphDatabase.driver(
            "bolt://localhost:7687",
            auth=("neo4j", "password")
        )
    def get_cases(self, query):
        return self.case_retriever.retrieve_similar_cases(query)
    def get_related_clauses(self, clause_types):
        with self.graph.session() as session:
            result = session.run("""
                MATCH (c:Clause)
                WHERE c.clause_type IN $types
                RETURN c.text AS clause_text
                LIMIT 10
            """, {"types": clause_types})
            return [r["clause_text"] for r in result]
    def get_contracts(self, clause_types):
        with self.graph.session() as session:
            result = session.run("""
                MATCH (contract:Contract)-[:HAS_CLAUSE]->(c:Clause)
                WHERE c.clause_type IN $types
                RETURN contract.contract_id AS id,
                       contract.contract_value AS value
                LIMIT 10
            """, {"types": clause_types})
            return [dict(r) for r in result]
    def retrieve_all(self, query, clause_types):
        return {
            "cases": self.get_cases(query),
            "clauses": self.get_related_clauses(clause_types),
            "contracts": self.get_contracts(clause_types)
        }

# graphrag/graph_traversal.py
from neo4j import GraphDatabase
class GraphTraversal:
    def __init__(self):
        self.driver = GraphDatabase.driver(
            "bolt://localhost:7687",
            auth=("neo4j", "password")
        )
    def get_event_impact(self, event_text):
        with self.driver.session() as session:
            # 1. Match relevant clauses
            clauses = session.run("""
                MATCH (c:Clause)
                WHERE toLower(c.clause_type) IN ["force majeure", "indemnity", "termination"]
                RETURN c.clause_type AS type, c.text AS text
                LIMIT 10
            """)
            clause_data = [dict(r) for r in clauses]
            # 2. Find contracts linked to those clauses
            contracts = session.run("""
                MATCH (contract:Contract)-[:HAS_CLAUSE]->(c:Clause)
                WHERE c.clause_type IN ["force majeure", "indemnity", "termination"]
                RETURN contract.contract_id AS id,
                       contract.contract_value AS value
                LIMIT 10
            """)
            contract_data = [dict(r) for r in contracts]
            # 3. Traverse risks
            risks = session.run("""
                MATCH (c:Clause)-[:INTRODUCES_RISK]->(r:Risk)
                RETURN r.type AS type, r.severity AS severity
                LIMIT 10
            """)
            risk_data = [dict(r) for r in risks]
        return {
            "clauses": clause_data,
            "contracts": contract_data,
            "risks": risk_data
        }

# crawler/async_scraper.py
import aiohttp
import asyncio
from bs4 import BeautifulSoup
class LegalCrawler:
    def __init__(self, urls):
        self.urls = urls
    async def fetch(self, session, url):
        try:
            async with session.get(url, timeout=10) as response:
                return await response.text()
        except Exception as e:
            print(f"Error fetching {url}: {e}")
            return None
    async def scrape(self):
        async with aiohttp.ClientSession() as session:
            tasks = [self.fetch(session, url) for url in self.urls]
            pages = await asyncio.gather(*tasks)
        results = []
        for html, url in zip(pages, self.urls):
            if html:
                soup = BeautifulSoup(html, "html.parser")
                title = soup.title.string if soup.title else ""
                paragraphs = " ".join([p.get_text() for p in soup.find_all("p")])
                results.append({
                    "url": url,
                    "title": title,
                    "content": paragraphs[:5000]
                })
        return results

# nlp/legal_extractor.py
import re
class LegalExtractor:
    def extract_entities(self, text):
        return {
            "event_type": self.detect_event(text),
            "jurisdiction": self.detect_jurisdiction(text),
            "clauses": self.extract_clauses(text),
            "case_refs": self.extract_case_refs(text)
        }
    def detect_event(self, text):
        if "regulation" in text.lower():
            return "Regulatory Change"
        if "court" in text.lower():
            return "Judgment"
        if "sanction" in text.lower():
            return "Sanction"
        return "General"
    def detect_jurisdiction(self, text):
        if "india" in text.lower():
            return "India"
        if "supreme court" in text.lower():
            return "India"
        return "Global"
    def extract_clauses(self, text):
        keywords = ["indemnity", "force majeure", "termination", "arbitration"]
        return [kw for kw in keywords if kw in text.lower()]
    def extract_case_refs(self, text):
        # Simple pattern for case citations
        pattern = r"[A-Z][a-z]+ vs [A-Z][a-z]+"
        return re.findall(pattern, text)

class Neo4jWriter:
    def __init__(self, uri, user, password):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))
    def write_event(self, event):
        with self.driver.session() as session:
            session.run("""
                MERGE (e:Event {event_id: $id})
                SET e.title = $title,
                    e.type = $type,
                    e.jurisdiction = $jurisdiction
            """, {
                "id": event["event_id"],
                "title": event["title"],
                "type": event["event_type"],
                "jurisdiction": event["jurisdiction"]
            })
            for clause in event["clauses"]:
                session.run("""
                    MERGE (c:Clause {clause_type: $clause})
                    WITH c
                    MATCH (e:Event {event_id: $event_id})
                    MERGE (e)-[:IMPACTS]->(c)
                """, {
                    "clause": clause,
                    "event_id": event["event_id"]
                })

# services/risk_engine.py
import random
class RiskEngine:
    def compute(self, text, cases):
        base = len(cases) * 0.2
        probability = min(1.0, base + random.uniform(0.1, 0.5))
        impact = len(text) * 0.05
        return {
            "probability": round(probability, 2),
            "impact": round(impact, 2),
            "risk_score": round(probability * impact, 2)
        }

# graph/neo4j_writer.py
from neo4j import GraphDatabase
class Neo4jWriter:
    def __init__(self, uri, user, password):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))
    def write_clause_analysis(self, contract_id, clause):
        with self.driver.session() as session:
            session.run("""
                MERGE (c:Contract {id: $contract_id})
                MERGE (cl:Clause {id: $clause_id})
                SET cl.text = $text,
                    cl.risk_score = $risk_score
                MERGE (c)-[:HAS_CLAUSE]->(cl)
            """, {
                "contract_id": contract_id,
                "clause_id": clause["clause_id"],
                "text": clause["text"],
                "risk_score": clause["risk"]["risk_score"]
            })
            for case in clause["cases"]:
                session.run("""
                    MERGE (cs:Case {id: $case_id})
                    SET cs.text = $text
                    WITH cs
                    MATCH (cl:Clause {id: $clause_id})
                    MERGE (cl)-[:SUPPORTED_BY]->(cs)
                """, {
                    "case_id": case["case_id"],
                    "text": case["text"],
                    "clause_id": clause["clause_id"]
                })

# services/bayesian_risk_engine.py
import numpy as np
class BayesianRiskEngine:
    def __init__(self):
        self.legal_cpt = LEGAL_RISK_CPT
        self.litigation_cpt = LITIGATION_CPT
        self.financial_cpt = FINANCIAL_RISK_CPT
    def infer_legal_risk(self, event, clause_strength, jurisdiction):
        key = (event, clause_strength, jurisdiction)
        probs = self.legal_cpt.get(key, [0.3, 0.4, 0.3])
        return self._sample(probs)
    def infer_litigation(self, legal_risk, counterparty):
        key = (legal_risk, counterparty)
        probs = self.litigation_cpt.get(key, [0.3, 0.4, 0.3])
        return self._sample(probs)
    def infer_financial_risk(self, litigation, exposure_level):
        key = (litigation, exposure_level)
        probs = self.financial_cpt.get(key, [0.3, 0.4, 0.3])
        return self._sample(probs)
    def compute(self, inputs):
        legal = self.infer_legal_risk(
            inputs["event"],
            inputs["clause_strength"],
            inputs["jurisdiction"]
        )
        litigation = self.infer_litigation(
            legal,
            inputs["counterparty"]
        )
        financial = self.infer_financial_risk(
            litigation,
            inputs["exposure"]
        )
        return {
            "legal_risk": legal,
            "litigation": litigation,
            "financial_risk": financial
        }
    def _sample(self, probs):
        states = ["Low", "Medium", "High"]
        return np.random.choice(states, p=probs)

# services/cpt_learner.py
from collections import defaultdict
class CPTLearner:
    def __init__(self):
        self.counts = defaultdict(lambda: [0, 0, 0])
    def train(self, data):
        for row in data:
            key = (
                row["event"],
                row["clause_strength"],
                row["jurisdiction"]
            )
            outcome = row["outcome"]["legal_risk"]
            idx = ["Low", "Medium", "High"].index(outcome)
            self.counts[key][idx] += 1
    def normalize(self):
        cpt = {}
        for k, v in self.counts.items():
            total = sum(v)
            cpt[k] = [x / total for x in v]
        return cpt
🧠 3. Bayesian Updating (Online Learning)
def update_cpt(existing, new_counts):
    updated = {}
    for k in existing:
        updated[k] = [
            (existing[k][i] + new_counts[k][i]) / 2
            for i in range(3)
        ]
    return updated

import requests
from bs4 import BeautifulSoup
from concurrent.futures import ThreadPoolExecutor
HEADERS = {"User-Agent": "Mozilla/5.0"}
def fetch(url):
    try:
        res = requests.get(url, headers=HEADERS, timeout=10)
        return res.text
    except:
        return None
def parse(html, selectors):
    soup = BeautifulSoup(html, "html.parser")
    texts = []
    for sel in selectors:
        for tag in soup.select(sel):
            text = tag.get_text(strip=True)
            if len(text) > 50:
                texts.append(text)
    return " ".join(texts)

def create_event_node(event, entities):
    with driver.session() as session:
        session.run("""
        MERGE (e:Event {name:$event})
        SET e.timestamp = timestamp()
        """, event=event)
        for ent in entities:
            session.run("""
            MERGE (x:Entity {name:$ent})
            MERGE (e:Event {name:$event})
            MERGE (e)-[:INVOLVES]->(x)
            """, ent=ent, event=event)

def run_simulation(event):
    impacted_clauses = []
    with driver.session() as session:
        result = session.run("""
        MATCH (e:Event {name:$event})-[:IMPACTS]->(c:Clause)
        RETURN c.text
        """, event=event)
        impacted_clauses = [r["c.text"] for r in result]
    outputs = []
    for clause in impacted_clauses:
        impact = simulate_impact(clause, event)
        outputs.append({
            "clause": clause,

        })
    return outputs

def fetch_graph_context(query):
    with driver.session() as session:
        result = session.run("""
        MATCH (c:Clause)
        WHERE c.text CONTAINS $query
        OPTIONAL MATCH (c)<-[:HAS_CLAUSE]-(contract)
        OPTIONAL MATCH (c)-[:CITED_IN]->(case)
        OPTIONAL MATCH (case)-[:DECIDED_BY]->(court)
        RETURN c.text, contract.name, case.name, court.name
        LIMIT 10
        """, query=query)
        return [r.data() for r in result]
🧠 6. CONTEXT FUSION ENGINE
def build_context(query):
    vector_results = search_similar(query)
    graph_results = fetch_graph_context(query)
    context = "\n\n---VECTOR RESULTS---\n"
    context += "\n".join(vector_results)
    context += "\n\n---GRAPH RESULTS---\n"
    for g in graph_results:
        context += str(g) + "\n"
    return context
