5. BERT Embedding Engine
Uses:
MiniLM
Legal-BERT
from sentence_transformers import SentenceTransformer
from transformers import AutoTokenizer, AutoModel
import torch

minilm = SentenceTransformer('all-MiniLM-L6-v2')

tokenizer = AutoTokenizer.from_pretrained(
    "nlpaueb/legal-bert-base-uncased"
)

legalbert = AutoModel.from_pretrained(
    "nlpaueb/legal-bert-base-uncased"
)

def embed_text(text):

    embedding1 = minilm.encode(text)

    inputs = tokenizer(text, return_tensors="pt", truncation=True)

    outputs = legalbert(**inputs)

    embedding2 = outputs.last_hidden_state.mean(dim=1)

    return torch.cat([torch.tensor(embedding1), embedding2.squeeze()])


6. RAG Engine (BM25 + BERT)
from rank_bm25 import BM25Okapi

class RAGEngine:

    def __init__(self, documents):
        self.docs = documents
        tokenized = [doc.split() for doc in documents]
        self.bm25 = BM25Okapi(tokenized)

    def retrieve(self, query):

        scores = self.bm25.get_scores(query.split())
        top_docs = sorted(
            zip(self.docs, scores),
            key=lambda x: x[1],
            reverse=True
        )[:5]

        return [doc for doc,score in top_docs]

import torch.nn as nn

class DisputePredictor(nn.Module):

    def __init__(self,input_dim):

        super().__init__()

        self.model = nn.Sequential(
            nn.Linear(input_dim,256),
            nn.ReLU(),
            nn.Linear(256,64),
            nn.ReLU(),
            nn.Linear(64,1),
            nn.Sigmoid()
        )

    def forward(self,x):
        return self.model(x)


8. Qwen2.5 Reasoning Layer
Used for explanations.
from transformers import AutoTokenizer, AutoModelForCausalLM

tokenizer = AutoTokenizer.from_pretrained(
    "Qwen/Qwen2.5-0.5B"
)

model = AutoModelForCausalLM.from_pretrained(
    "Qwen/Qwen2.5-0.5B"
)

def generate_explanation(contract_text,prob):

    prompt = f"""
    Contract:
    {contract_text}

    Predicted dispute probability: {prob}

    Explain key legal risks.
    """

    inputs = tokenizer(prompt, return_tensors="pt")

    output = model.generate(**inputs,max_new_tokens=200)

    return tokenizer.decode(output[0])


9. Prediction Pipeline
def predict_dispute(contract_text):

    embedding = embed_text(contract_text)

    prob = model(
        embedding.unsqueeze(0)
    ).item()

    explanation = generate_explanation(
        contract_text,
        prob
    )

    return prob, explanation

9. Graph Neural Network (Dispute Prediction)
Install:
pip install torch-geometric


GNN Model
import torch
import torch.nn.functional as F
from torch_geometric.nn import GCNConv

class DisputeGNN(torch.nn.Module):

    def __init__(self):

        super().__init__()

        self.conv1 = GCNConv(16,32)
        self.conv2 = GCNConv(32,16)

        self.linear = torch.nn.Linear(16,1)

    def forward(self,data):

        x,edge_index = data.x,data.edge_index

        x = self.conv1(x,edge_index)
        x = F.relu(x)

        x = self.conv2(x,edge_index)
        x = F.relu(x)

        x = self.linear(x)

        return torch.sigmoid(x)

This model learns risk propagation across the graph.

import numpy as np

class RiskNode:

    def __init__(self,state):

        self.state = state
        self.children = []
        self.visits = 0
        self.reward = 0


Selection
def select(node):

    best = None
    best_score = -1e9

    for child in node.children:

        ucb = (
            child.reward / (child.visits+1) +
            np.sqrt(np.log(node.visits+1)/(child.visits+1))
        )

        if ucb > best_score:
            best_score = ucb
            best = child

    return best


Simulation
def simulate(state):

    risk = (
        0.3*state["supplier_delay"] +
        0.2*state["commodity_shock"] +
        0.2*state["credit_risk"] +
        0.3*state["contract_ambiguity"]
    )

    dispute_prob = min(1,risk)

    return dispute_prob


MCTS Search
def mcts(root,iterations):

    for _ in range(iterations):

        node = select(root)

        reward = simulate(node.state)

        node.visits += 1
        node.reward += reward

7. Monte Carlo Tree Search
Each node = contract configuration.
class Node:

    def __init__(self,state):

        self.state = state
        self.children=[]
        self.visits=0
        self.value=0


Selection
import numpy as np

def select(node):

    best=None
    best_score=-1e9

    for child in node.children:

        score = child.value/(child.visits+1) + \
        np.sqrt(np.log(node.visits+1)/(child.visits+1))

        if score>best_score:
            best_score=score
            best=child

    return best


Expansion
def expand(node):

    actions = range(6)

    for a in actions:

        new_state=node.state.copy()
        new_state[a]+=1

        child=Node(new_state)
        node.children.append(child)


Simulation
def simulate(state):

    for i in range(5):

        action=np.random.randint(0,6)

        state[action]+=1

    return contract_score(state)


Backpropagation
def backprop(node,reward):

    while node:

        node.visits+=1
        node.value+=reward

        node=node.parent


8. Negotiation Loop
def negotiate(initial_state):

    root=Node(initial_state)

    for i in range(1000):

        node=select(root)

        expand(node)

        reward=simulate(node.state)

        backprop(node,reward)

    best=max(root.children,key=lambda c:c.value)

    return best.state

5. Negotiation Environment
class NegotiationEnv:

    def __init__(self,state):

        self.state = state

    def step(self,action):

        if action=="increase_price":
            self.state["price"] += 2

        if action=="decrease_delivery":
            self.state["delivery_days"] -= 1

        reward = contract_score(self.state)

        return self.state,reward


6. Contract Score Function
Score balances commercial value vs dispute risk.
def contract_score(state):

    commercial_value = (
        0.5 * state["price"]
        -0.2 * state["delivery_days"]
        -0.2 * state["termination_penalty"]
    )

    dispute = dispute_probability(state)

    return commercial_value - 100 * dispute


7. Dispute Probability Model
def dispute_probability(state):

    risk = (

        0.25*state["delivery_days"] +
        0.25*state["payment_terms"] +
        0.25*state["liability_cap"] +
        0.25*state["termination_penalty"]

    )

    return min(1,risk/100)

class Node:

    def __init__(self,state):

        self.state = state
        self.children = []
        self.visits = 0
        self.value = 0


Selection
def select(node):

    best=None
    best_score=-1e9

    for child in node.children:

        score = child.value/(child.visits+1)

        if score>best_score:
            best_score=score
            best=child

    return best

import {Slider} from "@mui/material"

from sentence_transformers import SentenceTransformer

model = SentenceTransformer("all-MiniLM-L6-v2")

def predict_risk(text):

embedding = model.encode(text)

return float(embedding.mean())


Dispute Predictor
def predict_dispute(contract):

risk = predict_risk(contract.text)

return min(1,risk*1.5)


Negotiation Tree API
@api_view(['GET'])

def negotiation_tree(request,contract_id):

nodes = []
edges = []

for s in NegotiationState.objects.filter(contract_id=contract_id):

nodes.append({
"id":s.id,
"data":{"label":s.score}
})

edges.append({
"source":s.parent,
"target":s.id
})

return Response({"nodes":nodes,"edges":edges})

4. Neo4j Query API (NodeJS Gateway)
// routes/graph.js

const neo4j = require("neo4j-driver")

const driver = neo4j.driver(
"bolt://localhost:7687",
neo4j.auth.basic("neo4j","password")
)

exports.getRiskGraph = async(req,res)=>{

const session = driver.session()

const result = await session.run(

`MATCH (c:Contract)-[r]->(n)
RETURN c,r,n LIMIT 200`

)

res.json(result.records)

}
